import { formatCount } from "@/lib/format";

const orders: number[] = [];

export async function POST(request: Request): Promise<Response> {
  const body = (await request.json()) as { cents: number };
  orders.push(body.cents);
  return Response.json({ id: "order_" + orders.length, cents: body.cents });
}

export function GET(): Response {
  return Response.json({ count: formatCount(orders.length) });
}
